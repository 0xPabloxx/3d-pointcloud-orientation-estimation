        draw=black!70,

        font=\small\bfseries,
        text=white,

        draw=black!70,
        fill=#1,
        minimum height=0.8cm,
        minimum width=0.8cm,
        rounded corners=3pt,
        font=\normalsize\bfseries,
        text=white,

    lossnode/.style={
        draw=lossred!80!black,
        fill=lossred,

        minimum size=1.2cm,

        text=white,
        line width=1pt,
    },
    arrow/.style={
        ->,
        >=Stealth,
        line width=0.6pt,
        color=black!60,
    },
    redarrow/.style={
        ->,
        >=Stealth,

    },
    label/.style={
        font=\scriptsize,
        text=black!70,
    },
    blocklabel/.style={
        font=\footnotesize\bfseries,
        text=white,
    },
}
